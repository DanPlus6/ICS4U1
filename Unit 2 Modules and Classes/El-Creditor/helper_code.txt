function tempArrayAndCountryArrayBuilder(){
    //create array for all countries and hashmap of string, key: country to string array, value: valid bank within country
    const allCountries: string[] = [];
    const Country_To_Bank: Map<string, string[]> = new Map<string, string[]>()

    //create CSV read stream, reading bin-list-data.csv
    fs.createReadStream('bin-list-data.csv')
    .pipe(csv())
    // for every row of data, we look at the isoCode2 and issuer
    .on('data', (row) =>{
        const {isoCode2, Issuer} = row
        //define exist variable with either type string array if it exists or undefined if it doesn't
        const exist: string[] | undefined = Country_To_Bank.get(isoCode2)
        // if the key value pair doesn't exist we define the KV (key value) pair we also filter for no name countries and banks
        if (!exist && Issuer != "" && isoCode2 != ""){
            Country_To_Bank.set(isoCode2, [Issuer])
        }
        // otherwise if a KV pair exists and we don't already have the specific issuer in the hashmap value array we add the issuer
        // into the string array also we filter for no name banks
        else if(exist && !(exist.includes(Issuer)) && Issuer != ""){
            exist.push(Issuer)
        }
        // if the country doesn't exist in the country array, we add the country to the string array we also filter for no name countries
        if(!(allCountries.includes(isoCode2)) && isoCode2 != ""){
            allCountries.push(isoCode2);
        }
    })
    //Write both array and hashmap into objects so we can recall and use them later
    .on("end", () => {
        const obj = Object.fromEntries(Country_To_Bank);
        fs.writeFileSync(
            "Banks.json",
            JSON.stringify(obj, null, 2),
            "utf-8"
        )
        fs.writeFileSync(
            "Countries.json",
            JSON.stringify(allCountries, null, 2),
            "utf-8"
        )
        console.log('I did the thing')
    })
}
